\section*{Core Setup}

\begin{lstlisting}[style=rstyle]
library(shiny)
library(bs4Dash)
library(bs4Dashkit)

ttl <- dash_titles(
  brand_text = "My Dashboard",
  icon = icon("cloud")
)

ui <- bs4DashPage(
  title = ttl$app_name,
  header = bs4DashNavbar(title = ttl$brand),
  sidebar = bs4DashSidebar(),
  body = bs4DashBody(
    use_bs4Dashkit_core(ttl)
  )
)
\end{lstlisting}

\textbf{Rule:} put \texttt{use\_bs4Dashkit\_core(ttl)} first inside
\texttt{bs4DashBody(...)}.

\section*{Brand Builder}

\texttt{dash\_titles()} returns:
\begin{itemize}
  \item \texttt{ttl\$app\_name}: browser title
  \item \texttt{ttl\$brand}: navbar or sidebar brand UI
  \item \texttt{ttl\$deps}: dependencies used by the core helper
\end{itemize}

\textbf{Icon inputs}
\begin{itemize}
  \item character name: \texttt{icon = "cloud"}
  \item Shiny tag: \texttt{icon = icon("cloud")}
  \item image logo: \texttt{icon\_img = "logo.png"}
\end{itemize}

\section*{Sidebar Brand Modes}

\begin{tabular}{>{\ttfamily}l l}
\toprule
Mode & Meaning \\
\midrule
icon-only & icon only \\
icon-text & icon + label \\
text-only & label only \\
\bottomrule
\end{tabular}

\begin{lstlisting}[style=rstyle]
ttl <- dash_titles(
  brand_text = "OLTCR Dashboards",
  icon = icon("cloud"),
  collapsed = "icon-text",
  expanded = "icon-text",
  collapsed_text = "OLTCR",
  expanded_text = "OLTCR Dashboards"
)
\end{lstlisting}

\textbf{New sidebar text styling}

\begin{lstlisting}[style=rstyle]
ttl <- dash_titles(
  brand_text = "OLTCR Dashboards",
  icon = icon("cloud"),
  collapsed_text_size = "10px",
  expanded_text_size = "15px",
  collapsed_text_weight = 700,
  expanded_text_weight = 800
)
\end{lstlisting}

\section*{Themes}

\textbf{Built-in presets}
\begin{itemize}
  \item \texttt{professional}
  \item \texttt{modern}
  \item \texttt{dark-lite}
\end{itemize}

\begin{lstlisting}[style=rstyle]
bs4dashkit_theme_presets()

use_bs4Dashkit_core(
  ttl,
  preset = "modern"
)
\end{lstlisting}

\textbf{Accent override}

\begin{lstlisting}[style=rstyle]
use_bs4Dashkit_core(
  ttl,
  preset = "professional",
  accent = "#2f6f8f"
)
\end{lstlisting}

\textbf{Direct theme control}

\begin{lstlisting}[style=rstyle]
use_dash_theme(
  bg = "#f5f6f8",
  surface = "#ffffff",
  accent = "#2f6f8f"
)
\end{lstlisting}
